Reading nested dictionaries is massively helped by using [[https://docs.python.org/3/library/pprint.html][pprint]].

#+begin_example python
from pprint import pp

nested = {"a": {"b": {"c": {"d": [1, 2, 3], "e": [4, 5, 6]}, "f": {"g": [7, 8, 9], "h": [10, 11, 12]}}}}

pp(nested, depth=5, compact=False)

{'a': {'b': {'c': {'d': [1, 2, 3], 'e': [4, 5, 6]},
             'f': {'g': [7, 8, 9], 'h': [10, 11, 12]}}}}
#+end_example
